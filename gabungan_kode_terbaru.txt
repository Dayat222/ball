/**
 * In-App OTA Update Manager for seamless Over-The-Air application updates.
 * Checks for updates via GitHub raw version.json (with GitHub API fallback),
 * downloads APK in background following HTTP redirects, and triggers system package installer.
 */
object AppUpdateManager {

    private const val VERSION_JSON_URL =
        "https://raw.githubusercontent.com/Dayat222/ball/main/version.json"
    private const val GITHUB_LATEST_RELEASE_API =
        "https://api.github.com/repos/Dayat222/ball/releases/latest"

    data class UpdateInfo(
        val remoteVersionCode: Long,
        val remoteVersionName: String,
        val downloadUrl: String,
        val changelog: String,
        val isUpdateAvailable: Boolean
    )

    fun getInstalledVersionCode(context: Context): Long {
        return try {
            val packageInfo = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                context.packageManager.getPackageInfo(
                    context.packageName,
                    PackageManager.PackageInfoFlags.of(0)
                )
            } else {
                @Suppress("DEPRECATION")
                context.packageManager.getPackageInfo(context.packageName, 0)
            }
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                packageInfo.longVersionCode
            } else {
                @Suppress("DEPRECATION")
                packageInfo.versionCode.toLong()
            }
        } catch (e: Exception) {
            1L
        }
    }

    fun getInstalledVersionName(context: Context): String {
        return try {
            val packageInfo = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                context.packageManager.getPackageInfo(
                    context.packageName,
                    PackageManager.PackageInfoFlags.of(0)
                )
            } else {
                @Suppress("DEPRECATION")
                context.packageManager.getPackageInfo(context.packageName, 0)
            }
            packageInfo.versionName ?: "1.0"
        } catch (e: Exception) {
            "1.0"
        }
    }

    /**
     * Checks if a newer version exists online.
     * Runs asynchronously on Dispatchers.IO and posts result back on Dispatchers.Main.
     */
    fun checkForUpdate(
        context: Context,
        onResult: (UpdateInfo) -> Unit,
        onError: (Exception) -> Unit
    ) {
        val currentVersionCode = getInstalledVersionCode(context)

        CoroutineScope(Dispatchers.IO).launch {
            try {
                var updateInfo = fetchFromVersionJson(currentVersionCode)
                if (updateInfo == null) {
                    // Fallback to GitHub Releases API if version.json is unreachable
                    updateInfo = fetchFromGitHubReleases(currentVersionCode)
                }

                if (updateInfo != null) {
                    withContext(Dispatchers.Main) {
                        onResult(updateInfo)
                    }
                } else {
                    withContext(Dispatchers.Main) {
                        onError(IllegalStateException("Gagal memuat informasi versi rilis."))
                    }
                }
            } catch (e: Exception) {
                withContext(Dispatchers.Main) {
                    onError(e)
                }
            }
        }
    }

    private fun fetchFromVersionJson(currentVersionCode: Long): UpdateInfo? {
        return try {
            val cacheBusterUrl = "$VERSION_JSON_URL?t=${System.currentTimeMillis()}"
            val conn = openConnectionWithRedirects(cacheBusterUrl)
            if (conn.responseCode != HttpURLConnection.HTTP_OK) {
                conn.disconnect()
                return null
            }
            val jsonText = conn.inputStream.bufferedReader().use { it.readText() }
            conn.disconnect()

            val json = JSONObject(jsonText)
            val remoteCode = json.optLong("versionCode", 1L)
            val remoteName = json.optString("versionName", "1.0")
            val downloadUrl = json.optString("downloadUrl", "")
            val changelog = json.optString("changelog", "Pembaruan kestabilan dan perbaikan fitur.")

            UpdateInfo(
                remoteVersionCode = remoteCode,
                remoteVersionName = remoteName,
                downloadUrl = downloadUrl,
                changelog = changelog,
                isUpdateAvailable = remoteCode > currentVersionCode
            )
        } catch (e: Exception) {
            null
        }
    }

    private fun fetchFromGitHubReleases(currentVersionCode: Long): UpdateInfo? {
        return try {
            val conn = openConnectionWithRedirects(GITHUB_LATEST_RELEASE_API)
            if (conn.responseCode != HttpURLConnection.HTTP_OK) {
                conn.disconnect()
                return null
            }
            val jsonText = conn.inputStream.bufferedReader().use { it.readText() }
            conn.disconnect()

            val json = JSONObject(jsonText)
            val tagName = json.optString("tag_name", "v1.0")
            val body = json.optString("body", "Pembaruan otomatis terbaru.")

            var apkUrl = ""
            val assets = json.optJSONArray("assets")
            if (assets != null) {
                for (i in 0 until assets.length()) {
                    val asset = assets.getJSONObject(i)
                    val name = asset.optString("name", "")
                    if (name.endsWith(".apk", ignoreCase = true)) {
                        apkUrl = asset.optString("browser_download_url", "")
                        break
                    }
                }
            }

            if (apkUrl.isEmpty()) {
                apkUrl = "https://github.com/Dayat222/ball/releases/download/latest/app-debug.apk"
            }

            // Estimate remote version code from tag if format is v1.2 or similar
            val tagNumber = tagName.removePrefix("v").replace(".", "").toLongOrNull() ?: (currentVersionCode + 1)

            UpdateInfo(
                remoteVersionCode = tagNumber,
                remoteVersionName = tagName,
                downloadUrl = apkUrl,
                changelog = body,
                isUpdateAvailable = true
            )
        } catch (e: Exception) {
            null
        }
    }

    /**
     * Downloads APK from updateInfo.downloadUrl with multi-hop redirect following,
     * reporting percentage progress (0-100), and then triggers installApk.
     */
    fun downloadAndInstall(
        context: Context,
        updateInfo: UpdateInfo,
        onProgress: (Int) -> Unit = {},
        onComplete: (File) -> Unit = {},
        onError: (Exception) -> Unit = {}
    ) {
        val destFile = File(context.cacheDir, "app-update.apk")

        CoroutineScope(Dispatchers.IO).launch {
            try {
                if (destFile.exists()) {
                    destFile.delete()
                }

                val conn = openConnectionWithRedirects(updateInfo.downloadUrl)
                if (conn.responseCode != HttpURLConnection.HTTP_OK) {
                    throw IllegalStateException("Server merespons kode status: ${conn.responseCode}")
                }

                val totalLength = conn.contentLength.toLong()
                var input: InputStream? = null
                var output: FileOutputStream? = null

                try {
                    input = conn.inputStream
                    output = FileOutputStream(destFile)

                    val buffer = ByteArray(8192)
                    var bytesRead: Int
                    var totalDownloaded = 0L
                    var lastReportedPercent = -1

                    while (input.read(buffer).also { bytesRead = it } != -1) {
                        output.write(buffer, 0, bytesRead)
                        totalDownloaded += bytesRead

                        if (totalLength > 0) {
                            val percent = ((totalDownloaded * 100) / totalLength).toInt().coerceIn(0, 100)
                            if (percent != lastReportedPercent) {
                                lastReportedPercent = percent
                                withContext(Dispatchers.Main) {
                                    onProgress(percent)
                                }
                            }
                        }
                    }
                    output.flush()
                } finally {
                    output?.close()
                    input?.close()
                    conn.disconnect()
                }

                if (!destFile.exists() || destFile.length() == 0L) {
                    throw IllegalStateException("File unduhan kosong atau rusak.")
                }

                destFile.setReadable(true, false)

                withContext(Dispatchers.Main) {
                    onComplete(destFile)
                    installApk(context, destFile)
                }
            } catch (e: Exception) {
                withContext(Dispatchers.Main) {
                    onError(e)
                }
            }
        }
    }

    /**
     * Launches the native Android package installer using FileProvider.
     * Directs user to grant Unknown App Sources permission if needed on Android 8.0+.
     */
    fun installApk(context: Context, apkFile: File) {
        if (!apkFile.exists() || apkFile.length() == 0L) {
            Toast.makeText(context, R.string.update_download_failed, Toast.LENGTH_SHORT).show()
            return
        }

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            if (!context.packageManager.canRequestPackageInstalls()) {
                val settingsIntent = Intent(Settings.ACTION_MANAGE_UNKNOWN_APP_SOURCES).apply {
                    data = Uri.parse("package:${context.packageName}")
                    addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                }
                context.startActivity(settingsIntent)
                Toast.makeText(
                    context,
                    "Izinkan 'Instal aplikasi tidak dikenal' lalu ulangi update",
                    Toast.LENGTH_LONG
                ).show()
                return
            }
        }

        try {
            val apkUri = FileProvider.getUriForFile(
                context,
                "${context.packageName}.fileprovider",
                apkFile
            )

            val installIntent = Intent(Intent.ACTION_VIEW).apply {
                setDataAndType(apkUri, "application/vnd.android.package-archive")
                addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
                addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            }
            context.startActivity(installIntent)
        } catch (e: Exception) {
            e.printStackTrace()
            Toast.makeText(context, "Gagal meluncurkan installer: ${e.message}", Toast.LENGTH_LONG).show()
        }
    }

    /**
     * Opens an HttpURLConnection, following 301/302/307/308 redirects up to 6 times.
     * Essential for GitHub Release downloads redirecting to AWS S3 storage.
     */
    private fun openConnectionWithRedirects(initialUrl: String): HttpURLConnection {
        var currentUrl = initialUrl
        var redirects = 0

        while (redirects < 6) {
            val conn = (URL(currentUrl).openConnection() as HttpURLConnection).apply {
                instanceFollowRedirects = true
                connectTimeout = 15000
                readTimeout = 15000
                setRequestProperty("User-Agent", "Mozilla/5.0 (Linux; Android 14)")
            }
            conn.connect()

            val status = conn.responseCode
            if (status == HttpURLConnection.HTTP_MOVED_TEMP ||
                status == HttpURLConnection.HTTP_MOVED_PERM ||
                status == HttpURLConnection.HTTP_SEE_OTHER ||
                status == 307 || status == 308
            ) {
                val newUrl = conn.getHeaderField("Location")
                conn.disconnect()
                if (newUrl.isNullOrEmpty()) {
                    throw IllegalStateException("Redirected with empty Location header")
                }
                currentUrl = newUrl
                redirects++
            } else {
                return conn
            }
        }
        throw IllegalStateException("Too many HTTP redirects from $initialUrl")
    }

    /**
     * Helper to show a clean Material Dialog on an Activity when an update is available.
     */
    fun showUpdateDialog(
        activity: Activity,
        updateInfo: UpdateInfo,
        onDownloadStarted: (() -> Unit)? = null
    ) {
        val message = "${updateInfo.changelog}\n\nVersi: ${updateInfo.remoteVersionName}"

        MaterialAlertDialogBuilder(activity)
            .setTitle(activity.getString(R.string.update_available_title))
            .setMessage(message)
            .setPositiveButton(activity.getString(R.string.update_now)) { _, _ ->
                onDownloadStarted?.invoke()
                val progressDialog = MaterialAlertDialogBuilder(activity)
                    .setTitle(activity.getString(R.string.update_checking))
                    .setMessage(activity.getString(R.string.update_downloading, 0))
                    .setCancelable(false)
                    .create()
                progressDialog.show()

                downloadAndInstall(
                    context = activity,
                    updateInfo = updateInfo,
                    onProgress = { percent ->
                        if (!activity.isFinishing && !activity.isDestroyed && progressDialog.isShowing) {
                            progressDialog.setMessage(activity.getString(R.string.update_downloading, percent))
                        }
                    },
                    onComplete = {
                        if (!activity.isFinishing && !activity.isDestroyed && progressDialog.isShowing) {
                            progressDialog.dismiss()
                        }
                    },
                    onError = { err ->
                        if (!activity.isFinishing && !activity.isDestroyed && progressDialog.isShowing) {
                            progressDialog.dismiss()
                        }
                        if (!activity.isFinishing && !activity.isDestroyed) {
                            Toast.makeText(
                                activity,
                                "${activity.getString(R.string.update_download_failed)}: ${err.message}",
                                Toast.LENGTH_LONG
                            ).show()
                        }
                    }
                )
            }
            .setNegativeButton(activity.getString(R.string.update_later), null)
            .show()
    }
}

    /**
     * Requirement 2: Auto-clamp boundaryBox to prevent overflowing screen edges in Landscape mode.
     */
    fun clampBoundaryBox(w: Float, h: Float) {
        if (w <= 0 || h <= 0) return
        val minMargin = 20f * density
        val minWidth = 80f * density
        val minHeight = 80f * density

        var l = boundaryBox.left.coerceIn(minMargin, (w * 0.45f).coerceAtLeast(minMargin))
        var t = boundaryBox.top.coerceIn(minMargin, (h * 0.45f).coerceAtLeast(minMargin))
        var r = boundaryBox.right.coerceIn((w * 0.55f).coerceAtMost(w - minMargin), w - minMargin)
        var b = boundaryBox.bottom.coerceIn((h * 0.55f).coerceAtMost(h - minMargin), h - minMargin)

        if (r <= l + minWidth) {
            l = minMargin
            r = w - minMargin
        }
        if (b <= t + minHeight) {
            t = minMargin
            b = h - minMargin
        }
        boundaryBox.set(l, t, r, b)
    }

    /**
     * Requirement 2: Reset boundary box to proportional center rectangle.
     */
    fun resetToProportionalCenter() {
        val landSize = getLandscapeScreenSize()
        val w = landSize.x.coerceAtLeast(200f)
        val h = landSize.y.coerceAtLeast(200f)
        val l = w * 0.191f
        val t = h * 0.228f
        val r = w * (1f - 0.191f)
        val b = h * 0.921f
        boundaryBox.set(l, t, r, b)
        prefsManager.saveBoundaryBox(l, t, r, b)
        invalidate()
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)
        val saveCount = canvas.save()
        if (!isCalibrating) {
            canvas.translate(-windowOffsetX, -windowOffsetY)
        }

        // Requirement 1: Clean View. ONLY draw boundary box and handles when isCalibrating == true
        if (isCalibrating) {
            drawBoundaryBox(canvas)
            drawEightCalibrationHandles(canvas)
        }

        if (!isCalibrating) {
            if (showRadialGuides) {
                // Requirement 2: Mode 1-Titik Fokus (Single-Point Target Mode)
                // Hide connecting line A-B and reflection rays.
                // Only render Point B's stem tether (from grip pad to Point B)
                if (hasGripB) {
                    canvas.drawLine(gripB.x, gripB.y, pointB.x, pointB.y, stemLinePaint)
                }
                // Render Point B's white center dot + 6 Direct rays + 6 Mirror Bank rays
                drawTargetAndRadialGuides(canvas)
            } else {
                // Normal modes (Auto-Reflect, Manual 3-Pt, Edge-Marker)
                drawStemTethers(canvas)
                when (currentMode) {
                    RulerMode.AUTO_REFLECT -> drawAutoReflection(canvas)
                    RulerMode.MANUAL_3_POINT -> drawManualThreePoint(canvas)
                    RulerMode.EDGE_MARKER_ONLY -> drawEdgeMarkerOnly(canvas)
                }
            }
        }

        canvas.restoreToCount(saveCount)
    }

    private fun drawStemTethers(canvas: Canvas) {

            canvas.drawCircle(pointA.x, pointA.y, 4f * density, aimDotPaint)
        }
        if (hasGripB) {

            canvas.drawCircle(pointC.x, pointC.y, 4f * density, aimDotPaint)
        }
    }

    /**
     * Zero-allocation helper to draw a radial ray from (bx, by) toward target (tx, ty).
     */
    private fun drawRadialRay(
        canvas: Canvas,
        bx: Float,
        by: Float,
        tx: Float,
        ty: Float,
        startDist: Float,
        endDist: Float,
        paint: Paint
    ) {
        val dx = tx - bx
        val dy = ty - by
        val dist = hypot(dx, dy)
        if (dist > 1f) {
            val ux = dx / dist
            val uy = dy / dist
            val startX = bx + ux * startDist
            val startY = by + uy * startDist
            val endX = bx + ux * endDist
            val endY = by + uy * endDist
            canvas.drawLine(startX, startY, endX, endY, paint)
        }
    }

    /**
     * Precision Lateral Vertex Ray (Top & Bottom Center Gate Clearance).
     * Accounts for approach tilt angle (kemiringan tembakan) across all cut angles.
     * In pool physics, as the shot becomes more angled (tiltRatio = |dx| / dist),
     * the near cushion knuckle physically obstructs the direct line of sight to the pocket center.
     * To prevent the ball from colliding with the near cushion knuckle and rattling out,
     * the entry target along the rail line MUST dynamically shift AWAY from the near knuckle
     * (towards the pocket opening center / far facing) based on ball radius and approach inclination.
     * Zero-allocation primitive Float calculations.
     */
    private fun drawLateralVertexRay(
        canvas: Canvas,
        bx: Float,
        by: Float,
        cx: Float,
        railY: Float,
        isTopGate: Boolean,
        startDist: Float,
        endDist: Float,
        paint: Paint
    ) {
        val dx = bx - cx
        val dy = if (isTopGate) (by - railY).coerceAtLeast(1f) else (railY - by).coerceAtLeast(1f)
        val dist = hypot(dx, dy)

        val ballR = ringRadiusDp * density
        val maxShift = (ballR * 0.60f).coerceIn(6f * density, 9f * density)

        // Tilt ratio ranges from 0.0 (straight-in) to 1.0 (parallel cut along cushion)
        val tiltRatio = if (dist > 1f) (abs(dx) / dist).coerceIn(0f, 1f) else 0f

        // Dynamic tilt compensation: shifts AWAY from the near cushion knuckle
        // (dx > 0 means ball is to the right, near knuckle is right, shift LEFT; vice-versa)
        val shiftMagnitude = (ballR * (tiltRatio * tiltRatio * 0.55f + tiltRatio * 0.15f)).coerceAtMost(maxShift)
        val shiftSign = if (dx > 0.5f) -1f else if (dx < -0.5f) 1f else 0f

        val targetX = cx + (shiftSign * shiftMagnitude)
        val targetY = railY

        val rayDx = targetX - bx
        val rayDy = targetY - by
        val rayDist = hypot(rayDx, rayDy)
        if (rayDist > 1f) {
            val ux = rayDx / rayDist
            val uy = rayDy / rayDist
            canvas.drawLine(
                bx + ux * startDist,
                by + uy * startDist,
                bx + ux * endDist,
                by + uy * endDist,
                paint
            )
        }
    }

    /**
     * Precision Corner Vertex Ray (4 Corner Gates: Top-Left, Top-Right, Bottom-Left, Bottom-Right).
     * Accounts for physical corner pocket mouth center, ensuring all rays accurately target the
     * corner pocket openings across any position on the table without clipping cushions.
     * Zero-allocation primitive Float calculations.
     */
    private fun drawCornerVertexRay(
        canvas: Canvas,
        bx: Float,
        by: Float,
        cornerX: Float,
        cornerY: Float,
        isLeft: Boolean,
        isTop: Boolean,
        startDist: Float,
        endDist: Float,
        paint: Paint
    ) {
        val signX = if (isLeft) 1f else -1f
        val signY = if (isTop) 1f else -1f

        // Precision corner pocket target center (titik pusat arah 4 lubang sudut)
        // Offset horizontal (9.0dp) dan vertikal (11.2dp) tepat mengarah ke titik pusat mulut lubang kantong
        val cornerMouthInsetX = 9.0f * density
        val cornerMouthInsetY = 11.2f * density
        val targetX = cornerX + (signX * cornerMouthInsetX)
        val targetY = cornerY + (signY * cornerMouthInsetY)

        val rayDx = targetX - bx
        val rayDy = targetY - by
        val rayDist = hypot(rayDx, rayDy)
        if (rayDist > 1f) {
            val ux = rayDx / rayDist
            val uy = rayDy / rayDist
            canvas.drawLine(
                bx + ux * startDist,
                by + uy * startDist,
                bx + ux * endDist,
                by + uy * endDist,
                paint
            )
        }
    }

    /**
     * Zero-Allocation Target Mode: 6 Direct Rays with 12-Jaw Gate Clearance + 10 Cushion Bank Rays.
     * Direct primitive calculations per frame without allocating any Array or PointF objects.
     */
    private fun drawTargetAndRadialGuides(canvas: Canvas) {
        val bx = pointB.x
        val by = pointB.y

        // 1. Ball-Fit Ring: High-Contrast Hollow Reticle matching ball size
        val ringRadius = ringRadiusDp * density
        canvas.drawCircle(bx, by, ringRadius, aimShadowPaint)
        canvas.drawCircle(bx, by, ringRadius, aimRingPaint)

        // 2. Micro Center Pinpoint: Double contrast micro center dot
        canvas.drawCircle(bx, by, 2.5f * density, aimCenterShadowPaint)
        canvas.drawCircle(bx, by, 1.5f * density, aimCenterDotPaint)

        // Draw the 6 direct rays and 10 mirror reflection rays if enabled
        if (showRadialGuides) {
            val startDist = ringRadius + (3f * density)
            val directEndDist = 85f * density
            val bankEndDist = 64f * density

            val l = boundaryBox.left
            val r = boundaryBox.right
            val t = boundaryBox.top
            val b = boundaryBox.bottom
            val cx = boundaryBox.centerX()

            val cornerJawDist = 14f * density

            // 1. 6 Direct Vertex Rays (4 Corner Gates with Rail Clearance + 2 Precision Lateral Gates with Tilt Modeling)
            // Top-Left (Corner)
            drawCornerVertexRay(canvas, bx, by, l, t, isLeft = true, isTop = true, startDist, directEndDist, sixLinePaint)

            // Top-Center (Precision Lateral Gate with Dynamic Tilt Angle Modeling - UNTOUCHED)
            drawLateralVertexRay(canvas, bx, by, cx, t, isTopGate = true, startDist, directEndDist, sixLinePaint)

            // Top-Right (Corner)
            drawCornerVertexRay(canvas, bx, by, r, t, isLeft = false, isTop = true, startDist, directEndDist, sixLinePaint)

            // Bottom-Left (Corner)
            drawCornerVertexRay(canvas, bx, by, l, b, isLeft = true, isTop = false, startDist, directEndDist, sixLinePaint)

            // Bottom-Center (Precision Lateral Gate with Dynamic Tilt Angle Modeling - UNTOUCHED)
            drawLateralVertexRay(canvas, bx, by, cx, b, isTopGate = false, startDist, directEndDist, sixLinePaint)

            // Bottom-Right (Corner)
            drawCornerVertexRay(canvas, bx, by, r, b, isLeft = false, isTop = false, startDist, directEndDist, sixLinePaint)

            // 2. 10 Mirror Bank Rays (Pantulan Dinding Rel)
            // Collision plane accounts for physical cushion nose + ball radius
            val cushionBounceInset = (5f * density) + (ringRadiusDp * density)
            val bounceL = l + cushionBounceInset
            val bounceR = r - cushionBounceInset
            val bounceT = t + cushionBounceInset
            val bounceB = b - cushionBounceInset

            val sideJawInset = 6f * density
            val mirrorTopCornerY = 2f * bounceT - b
            val mirrorTopCenterY = 2f * bounceT - (b - sideJawInset)
            val mirrorBottomCornerY = 2f * bounceB - t
            val mirrorBottomCenterY = 2f * bounceB - (t + sideJawInset)
            val mirrorRightX = 2f * bounceR - l
            val mirrorLeftX = 2f * bounceL - r

            val minClearance = 3f * density

            // Top Bound (3 rays: bounce off top rail toward bottom pockets)
            if (by > bounceT + minClearance) {
                drawRadialRay(canvas, bx, by, l + cornerJawDist, mirrorTopCornerY, startDist, bankEndDist, bankLinePaint)
                drawRadialRay(canvas, bx, by, cx, mirrorTopCenterY, startDist, bankEndDist, bankLinePaint)
                drawRadialRay(canvas, bx, by, r - cornerJawDist, mirrorTopCornerY, startDist, bankEndDist, bankLinePaint)
            }

            // Bottom Bound (3 rays: bounce off bottom rail toward top pockets)
            if (by < bounceB - minClearance) {
                drawRadialRay(canvas, bx, by, l + cornerJawDist, mirrorBottomCornerY, startDist, bankEndDist, bankLinePaint)
                drawRadialRay(canvas, bx, by, cx, mirrorBottomCenterY, startDist, bankEndDist, bankLinePaint)
                drawRadialRay(canvas, bx, by, r - cornerJawDist, mirrorBottomCornerY, startDist, bankEndDist, bankLinePaint)
            }

            // Right Bound (2 rays: bounce off right rail toward left pockets)
            if (bx < bounceR - minClearance) {
                drawRadialRay(canvas, bx, by, mirrorRightX, t + cornerJawDist, startDist, bankEndDist, bankLinePaint)
                drawRadialRay(canvas, bx, by, mirrorRightX, b - cornerJawDist, startDist, bankEndDist, bankLinePaint)
            }

            // Left Bound (2 rays: bounce off left rail toward right pockets)
            if (bx > bounceL + minClearance) {
                drawRadialRay(canvas, bx, by, mirrorLeftX, t + cornerJawDist, startDist, bankEndDist, bankLinePaint)
                drawRadialRay(canvas, bx, by, mirrorLeftX, b - cornerJawDist, startDist, bankEndDist, bankLinePaint)
            }
        }
    }

    private fun drawBoundaryBox(canvas: Canvas) {
        canvas.drawRect(boundaryBox, boundaryCalibPaint)

        // Feature 1: 6 Pocket Mouth Target Rings (Jaw Guides in Calibration Mode)
        val l = boundaryBox.left
        val r = boundaryBox.right
        val t = boundaryBox.top
        val b = boundaryBox.bottom
        val cx = boundaryBox.centerX()

        val cornerMouthInsetX = 9.0f * density
        val cornerMouthInsetY = 11.2f * density

        drawVertexGuideRing(canvas, l + cornerMouthInsetX, t + cornerMouthInsetY)   // Top-Left
        drawVertexGuideRing(canvas, cx, t)                                         // Top-Center
        drawVertexGuideRing(canvas, r - cornerMouthInsetX, t + cornerMouthInsetY)   // Top-Right
        drawVertexGuideRing(canvas, l + cornerMouthInsetX, b - cornerMouthInsetY)   // Bottom-Left
        drawVertexGuideRing(canvas, cx, b)                                         // Bottom-Center
        drawVertexGuideRing(canvas, r - cornerMouthInsetX, b - cornerMouthInsetY)   // Bottom-Right

        val label = "Calibration: Drag 8 handles to align target bounds"
        textPaint.getTextBounds(label, 0, label.length, textBounds)
        val textX = boundaryBox.left + 16f * density
        val textY = (boundaryBox.top - 12f * density).coerceAtLeast(36f * density)
        val padding = 6f * density
        canvas.drawRect(
            textX - padding,
            textY - textBounds.height() - padding,
            textX + textBounds.width() + padding,
            textY + padding,
            textBgPaint
        )
        canvas.drawText(label, textX, textY, textPaint)
    }

    private fun drawVertexGuideRing(canvas: Canvas, px: Float, py: Float) {
        val ringRadius = 12f * density
        canvas.drawCircle(px, py, ringRadius, vertexRingPaint)
        val crossArm = 4f * density
        canvas.drawLine(px - crossArm, py, px + crossArm, py, vertexCrossPaint)
        canvas.drawLine(px, py - crossArm, px, py + crossArm, vertexCrossPaint)
    }

    /**
     * Requirement 2: Render side-rail calibration handles (left and right edges)
     * and top/bottom center indicators.
     * The 6 pocket positions are clearly indicated by the 6 cyan jaw target rings.
     */
    private fun drawEightCalibrationHandles(canvas: Canvas) {
        val cx = boundaryBox.centerX()
        val cy = boundaryBox.centerY()
        val sideRadius = edgeHandleRadius * 0.75f

        // 2 Side Edge Centers (Left & Right rail resize handles)
        drawHandlePin(canvas, boundaryBox.left, cy, edgeHandleRadius)
        drawHandlePin(canvas, boundaryBox.right, cy, edgeHandleRadius)

        // Top & Bottom Edge Centers (Resize indicators)
        drawHandlePin(canvas, cx, boundaryBox.top, sideRadius)
        drawHandlePin(canvas, cx, boundaryBox.bottom, sideRadius)
    }

    private fun drawHandlePin(canvas: Canvas, cx: Float, cy: Float, radius: Float) {
        canvas.drawCircle(cx, cy, radius, handleOuterPaint)
        canvas.drawCircle(cx, cy, radius, handleStrokePaint)
        canvas.drawCircle(cx, cy, 4f * density, handleCenterPaint)
    }

    private fun updateInnerInsetBox() {
        val cushionBounceInset = (5f * density) + (ringRadiusDp * density)
        innerInsetBox.set(
            boundaryBox.left + cushionBounceInset,
            boundaryBox.top + cushionBounceInset,
            boundaryBox.right - cushionBounceInset,
            boundaryBox.bottom - cushionBounceInset
        )
    }

    private fun drawAutoReflection(canvas: Canvas) {
        updateInnerInsetBox()
        val reflection = GeometryUtils.computeReflection(pointA, pointB, innerInsetBox)

        if (reflection != null) {
            // Primary Line: A -> Hit Point
            linePath.reset()
            linePath.moveTo(pointA.x, pointA.y)
            linePath.lineTo(reflection.hitPoint.x, reflection.hitPoint.y)
            canvas.drawPath(linePath, primaryLinePaint)

            // Reflected Line: Hit Point -> Reflected End Point
            linePath.reset()
            linePath.moveTo(reflection.hitPoint.x, reflection.hitPoint.y)
            linePath.lineTo(reflection.reflectedPoint.x, reflection.reflectedPoint.y)
            canvas.drawPath(linePath, reflectedLinePaint)

            // Collision ring at boundary hit point
            canvas.drawCircle(
                reflection.hitPoint.x,
                reflection.hitPoint.y,
                6f * density,
                edgeMarkerPaint
            )
            canvas.drawCircle(
                reflection.hitPoint.x,
                reflection.hitPoint.y,
                6f * density,
                edgeMarkerStrokePaint
            )

            // Angle text badge near reflection point
            val angleText = String.format(Locale.US, "θ = %.1f°", reflection.angleOfIncidenceDeg)
            drawTextBadge(canvas, angleText, reflection.hitPoint.x + 12f * density, reflection.hitPoint.y - 12f * density)
        } else {
            // Fallback straight ray through B if no boundary hit
            linePath.reset()
            linePath.moveTo(pointA.x, pointA.y)
            linePath.lineTo(pointB.x, pointB.y)
            canvas.drawPath(linePath, primaryLinePaint)
        }
    }

     */
    @SuppressLint("ClickableViewAccessibility")
    override fun onTouchEvent(event: MotionEvent): Boolean {
        if (!isCalibrating) {
            return false
        }

        val touchX = event.x
        val touchY = event.y

        when (event.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                activeDragTarget = findTouchedCalibrationTarget(touchX, touchY)
                if (activeDragTarget != DragTarget.NONE) {
                    val targetPos = getTargetPos(activeDragTarget)
                    touchOffset.set(touchX - targetPos.x, touchY - targetPos.y)
                    return true
                }
                return false
            }

            MotionEvent.ACTION_MOVE -> {
                if (activeDragTarget != DragTarget.NONE) {
                    val minSize = 60f * density
                    val landSize = getLandscapeScreenSize()
                    val screenW = landSize.x
                    val screenH = landSize.y

                    val curX = (touchX - touchOffset.x).coerceIn(0f, screenW)
                    val curY = (touchY - touchOffset.y).coerceIn(0f, screenH)

                    when (activeDragTarget) {
                        DragTarget.CORNER_TOP_LEFT -> {
                            boundaryBox.left = curX.coerceAtMost(boundaryBox.right - minSize)
                            boundaryBox.top = curY.coerceAtMost(boundaryBox.bottom - minSize)
                        }
                        DragTarget.CORNER_TOP_RIGHT -> {
                            boundaryBox.right = curX.coerceAtLeast(boundaryBox.left + minSize)
                            boundaryBox.top = curY.coerceAtMost(boundaryBox.bottom - minSize)
                        }
                        DragTarget.CORNER_BOTTOM_LEFT -> {
                            boundaryBox.left = curX.coerceAtMost(boundaryBox.right - minSize)
                            boundaryBox.bottom = curY.coerceAtLeast(boundaryBox.top + minSize)
                        }
                        DragTarget.CORNER_BOTTOM_RIGHT -> {
                            boundaryBox.right = curX.coerceAtLeast(boundaryBox.left + minSize)
                            boundaryBox.bottom = curY.coerceAtLeast(boundaryBox.top + minSize)
                        }
                        DragTarget.HANDLE_TOP_CENTER -> {
                            boundaryBox.top = curY.coerceAtMost(boundaryBox.bottom - minSize)
                        }
                        DragTarget.HANDLE_BOTTOM_CENTER -> {
                            boundaryBox.bottom = curY.coerceAtLeast(boundaryBox.top + minSize)
                        }
                        DragTarget.HANDLE_LEFT_CENTER -> {
                            boundaryBox.left = curX.coerceAtMost(boundaryBox.right - minSize)
                        }
                        DragTarget.HANDLE_RIGHT_CENTER -> {
                            boundaryBox.right = curX.coerceAtLeast(boundaryBox.left + minSize)
                        }
                        DragTarget.NONE -> Unit
                    }
                    invalidate()
                    return true
                }
            }

            MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                if (activeDragTarget != DragTarget.NONE) {
                    activeDragTarget = DragTarget.NONE
                    prefsManager.saveBoundaryBox(boundaryBox.left, boundaryBox.top, boundaryBox.right, boundaryBox.bottom)
                    invalidate()
                    return true
                }
            }
        }

        return super.onTouchEvent(event)
    }

    private fun findTouchedCalibrationTarget(x: Float, y: Float): DragTarget {
        val cornerThreshold = 40f * density
        val edgeThreshold = 36f * density

        val cx = boundaryBox.centerX()
        val cy = boundaryBox.centerY()

        // 1. Check 4 Corners
        if (GeometryUtils.distance(x, y, boundaryBox.left, boundaryBox.top) <= cornerThreshold) {
            return DragTarget.CORNER_TOP_LEFT
        }
        if (GeometryUtils.distance(x, y, boundaryBox.right, boundaryBox.top) <= cornerThreshold) {
            return DragTarget.CORNER_TOP_RIGHT
        }
        if (GeometryUtils.distance(x, y, boundaryBox.left, boundaryBox.bottom) <= cornerThreshold) {
            return DragTarget.CORNER_BOTTOM_LEFT
        }
        if (GeometryUtils.distance(x, y, boundaryBox.right, boundaryBox.bottom) <= cornerThreshold) {
            return DragTarget.CORNER_BOTTOM_RIGHT
        }

        // 2. Check 4 Edge Centers
        if (GeometryUtils.distance(x, y, cx, boundaryBox.top) <= edgeThreshold) {
            return DragTarget.HANDLE_TOP_CENTER
        }
        if (GeometryUtils.distance(x, y, cx, boundaryBox.bottom) <= edgeThreshold) {
            return DragTarget.HANDLE_BOTTOM_CENTER
        }
        if (GeometryUtils.distance(x, y, boundaryBox.left, cy) <= edgeThreshold) {
            return DragTarget.HANDLE_LEFT_CENTER
        }
        if (GeometryUtils.distance(x, y, boundaryBox.right, cy) <= edgeThreshold) {
            return DragTarget.HANDLE_RIGHT_CENTER
        }

        return DragTarget.NONE
    }

    private fun getTargetPos(target: DragTarget): PointF {
        val cx = boundaryBox.centerX()
        val cy = boundaryBox.centerY()
        return when (target) {
            DragTarget.CORNER_TOP_LEFT -> PointF(boundaryBox.left, boundaryBox.top)
            DragTarget.CORNER_TOP_RIGHT -> PointF(boundaryBox.right, boundaryBox.top)
            DragTarget.CORNER_BOTTOM_LEFT -> PointF(boundaryBox.left, boundaryBox.bottom)
            DragTarget.CORNER_BOTTOM_RIGHT -> PointF(boundaryBox.right, boundaryBox.bottom)
            DragTarget.HANDLE_TOP_CENTER -> PointF(cx, boundaryBox.top)
            DragTarget.HANDLE_BOTTOM_CENTER -> PointF(cx, boundaryBox.bottom)
            DragTarget.HANDLE_LEFT_CENTER -> PointF(boundaryBox.left, cy)
            DragTarget.HANDLE_RIGHT_CENTER -> PointF(boundaryBox.right, cy)
            DragTarget.NONE -> PointF(0f, 0f)
        }
    }

package com.utility.screenprotractor

import android.graphics.PointF
import android.graphics.RectF
import kotlin.math.acos
import kotlin.math.atan2
import kotlin.math.sqrt

/**
 * Geometric calculation engine for ray-box intersections, vector reflections,
 * and 2D angular projections.
 */
object GeometryUtils {

    enum class BoundarySide {
        NONE, TOP, BOTTOM, LEFT, RIGHT
    }

    data class Intersection(
        val point: PointF,
        val side: BoundarySide,
        val distance: Float
    )

    data class ReflectionPath(
        val startPoint: PointF,
        val hitPoint: PointF,
        val hitSide: BoundarySide,
        val reflectedPoint: PointF,
        val angleOfIncidenceDeg: Float
    )

    /**
     * Compute Euclidean distance between two points.
     */
    fun distance(x1: Float, y1: Float, x2: Float, y2: Float): Float {
        val dx = x2 - x1
        val dy = y2 - y1
        return sqrt(dx * dx + dy * dy)
    }

    fun distance(p1: PointF, p2: PointF): Float {
        return distance(p1.x, p1.y, p2.x, p2.y)
    }

    /**
     * Calculate ray-box intersection from [origin] in direction of [directionPoint]
     * hitting the 4 edges of [box].
     *
     * @param origin Starting point of ray (Point A)
     * @param directionPoint Aiming point (Point B)
     * @param box Boundary rectangle
     * @param minDistance Minimum forward distance to avoid immediate self-collision
     * @return Intersection result with hit point, hit side, and distance, or null if no hit
     */
    fun computeRayBoxIntersection(
        origin: PointF,
        directionPoint: PointF,
        box: RectF,
        minDistance: Float = 0.5f
    ): Intersection? {
        val rawDx = directionPoint.x - origin.x
        val rawDy = directionPoint.y - origin.y
        val len = sqrt(rawDx * rawDx + rawDy * rawDy)

        if (len < 1e-5f) {
            return null
        }

        // Unit vector
        val ux = rawDx / len
        val uy = rawDy / len

        var closestT = Float.MAX_VALUE
        var hitSide = BoundarySide.NONE
        var hitX = 0f
        var hitY = 0f

        // 1. Check Horizontal Top Edge: y = box.top
        if (uy < -1e-5f) { // Moving upwards
            val t = (box.top - origin.y) / uy
            if (t > minDistance && t < closestT) {
                val x = origin.x + t * ux
                if (x in (box.left - 1f)..(box.right + 1f)) {
                    closestT = t
                    hitSide = BoundarySide.TOP
                    hitX = x.coerceIn(box.left, box.right)
                    hitY = box.top
                }
            }
        }

        // 2. Check Horizontal Bottom Edge: y = box.bottom
        if (uy > 1e-5f) { // Moving downwards
            val t = (box.bottom - origin.y) / uy
            if (t > minDistance && t < closestT) {
                val x = origin.x + t * ux
                if (x in (box.left - 1f)..(box.right + 1f)) {
                    closestT = t
                    hitSide = BoundarySide.BOTTOM
                    hitX = x.coerceIn(box.left, box.right)
                    hitY = box.bottom
                }
            }
        }

        // 3. Check Vertical Left Edge: x = box.left
        if (ux < -1e-5f) { // Moving leftwards
            val t = (box.left - origin.x) / ux
            if (t > minDistance && t < closestT) {
                val y = origin.y + t * uy
                if (y in (box.top - 1f)..(box.bottom + 1f)) {
                    closestT = t
                    hitSide = BoundarySide.LEFT
                    hitX = box.left
                    hitY = y.coerceIn(box.top, box.bottom)
                }
            }
        }

        // 4. Check Vertical Right Edge: x = box.right
        if (ux > 1e-5f) { // Moving rightwards
            val t = (box.right - origin.x) / ux
            if (t > minDistance && t < closestT) {
                val y = origin.y + t * uy
                if (y in (box.top - 1f)..(box.bottom + 1f)) {
                    closestT = t
                    hitSide = BoundarySide.RIGHT
                    hitX = box.right
                    hitY = y.coerceIn(box.top, box.bottom)
                }
            }
        }

        return if (hitSide != BoundarySide.NONE && closestT < Float.MAX_VALUE) {
            Intersection(PointF(hitX, hitY), hitSide, closestT)
        } else {
            null
        }
    }

    /**
     * Compute full reflection path:
     * 1. Primary ray from [pointA] towards [pointB] hitting [box] at [hitPoint].
     * 2. Reflected vector based on hit edge:
     *    - Horizontal (Top/Bottom): (dx', dy') = (dx, -dy)
     *    - Vertical (Left/Right): (dx', dy') = (-dx, dy)
     * 3. Reflected ray continues to the next boundary wall.
     */
    fun computeReflection(
        pointA: PointF,
        pointB: PointF,
        box: RectF
    ): ReflectionPath? {
        val firstHit = computeRayBoxIntersection(pointA, pointB, box) ?: return null

        val rawDx = pointB.x - pointA.x
        val rawDy = pointB.y - pointA.y
        val len = sqrt(rawDx * rawDx + rawDy * rawDy)
        if (len < 1e-5f) return null

        val ux = rawDx / len
        val uy = rawDy / len

        // Invert vector according to reflection surface normal
        val (refUx, refUy) = when (firstHit.side) {
            BoundarySide.TOP, BoundarySide.BOTTOM -> Pair(ux, -uy)
            BoundarySide.LEFT, BoundarySide.RIGHT -> Pair(-ux, uy)
            BoundarySide.NONE -> Pair(ux, uy)
        }

        // Calculate angle of incidence (with surface normal, between 0° and 90°)
        val normalX: Float
        val normalY: Float
        when (firstHit.side) {
            BoundarySide.TOP -> { normalX = 0f; normalY = 1f }
            BoundarySide.BOTTOM -> { normalX = 0f; normalY = -1f }
            BoundarySide.LEFT -> { normalX = 1f; normalY = 0f }
            BoundarySide.RIGHT -> { normalX = -1f; normalY = 0f }
            BoundarySide.NONE -> { normalX = 0f; normalY = 0f }
        }

        // Vector from hit point back towards incoming ray
        val incomingRevX = -ux
        val incomingRevY = -uy
        val dot = (incomingRevX * normalX + incomingRevY * normalY).coerceIn(-1f, 1f)
        val angleDeg = Math.toDegrees(acos(dot).toDouble()).toFloat()

        // Calculate bounce end point by intersecting reflected ray with next boundary wall
        val reflectedAimPoint = PointF(
            firstHit.point.x + refUx * 100f,
            firstHit.point.y + refUy * 100f
        )

        val secondHit = computeRayBoxIntersection(
            firstHit.point,
            reflectedAimPoint,
            box,
            minDistance = 2.0f
        )

        val bounceEnd = secondHit?.point ?: PointF(
            firstHit.point.x + refUx * 300f,
            firstHit.point.y + refUy * 300f
        )

        return ReflectionPath(
            startPoint = pointA,
            hitPoint = firstHit.point,
            hitSide = firstHit.side,
            reflectedPoint = bounceEnd,
            angleOfIncidenceDeg = angleDeg
        )
    }

    /**
     * Compute angle between two lines: (A -> B) and (B -> C) in degrees.
     */
    fun computeAngleBetweenLines(a: PointF, b: PointF, c: PointF): Float {
        val angle1 = atan2(a.y - b.y, a.x - b.x)
        val angle2 = atan2(c.y - b.y, c.x - b.x)
        var diff = Math.toDegrees((angle2 - angle1).toDouble()).toFloat()
        if (diff < 0) diff += 360f
        return diff
    }
}

    override fun onCreate() {
        super.onCreate()
        isServiceRunning = true

        windowManager = getSystemService(Context.WINDOW_SERVICE) as WindowManager
        prefsManager = PrefsManager(this)

        val density = resources.displayMetrics.density
        handleSizePx = (42f * density).roundToInt()
        handleOffsetX = 48f * density
        handleOffsetY = 36f * density

        // 1. Promote to Foreground Service
        startForegroundServiceWithNotification()

        // 2. Initialize and attach Canvas Overlay View
        setupCanvasOverlay()

        // 3. Initialize and attach 3 Independent Floating Handles with Anti-Finger Offset
        setupFloatingHandles()

        // 4. Initialize and attach Floating Control View (with Collapsible Minimize)
        setupFloatingControl()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        if (intent?.action == ACTION_STOP_SERVICE) {
            stopSelf()
            return START_NOT_STICKY
        }
        return START_STICKY
    }

    private fun startForegroundServiceWithNotification() {
        createNotificationChannel()

        val openAppIntent = Intent(this, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_SINGLE_TOP or Intent.FLAG_ACTIVITY_CLEAR_TOP
        }
        val openAppPendingIntent = PendingIntent.getActivity(
            this,
            0,
            openAppIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val stopIntent = Intent(this, OverlayService::class.java).apply {
            action = ACTION_STOP_SERVICE
        }
        val stopPendingIntent = PendingIntent.getService(
            this,
            1,
            stopIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val notification: Notification = NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle(getString(R.string.notification_title))
            .setContentText(getString(R.string.notification_text))
            .setSmallIcon(R.drawable.ic_notification)
            .setContentIntent(openAppPendingIntent)
            .setOngoing(true)
            .setPriority(NotificationCompat.PRIORITY_LOW)
            .addAction(
                R.drawable.ic_notification,
                getString(R.string.notification_action_stop),
                stopPendingIntent
            )
            .build()

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.UPSIDE_DOWN_CAKE) {
            startForeground(
                NOTIFICATION_ID,
                notification,
                ServiceInfo.FOREGROUND_SERVICE_TYPE_SPECIAL_USE
            )
        } else {
            startForeground(NOTIFICATION_ID, notification)
        }
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                getString(R.string.notification_channel_name),
                NotificationManager.IMPORTANCE_LOW
            ).apply {
                description = getString(R.string.notification_channel_desc)
                setShowBadge(false)
            }
            val manager = getSystemService(NotificationManager::class.java)
            manager?.createNotificationChannel(channel)
        }
    }

    private fun getOverlayWindowType(): Int {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
        } else {
            @Suppress("DEPRECATION")
            WindowManager.LayoutParams.TYPE_PHONE
        }
    }

    private fun getRealScreenSize(): Point {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
            val bounds = windowManager.currentWindowMetrics.bounds
            Point(bounds.width(), bounds.height())
        } else {
            val point = Point()
            @Suppress("DEPRECATION")
            windowManager.defaultDisplay.getRealSize(point)
            point
        }
    }

    private fun setupCanvasOverlay() {
        canvasOverlayView = CanvasOverlayView(this).apply {
            onCanvasSizeChanged = { _, _ ->
                repositionHandlesToAimPoints()
            }
        }

        canvasLayoutParams = WindowManager.LayoutParams(
            WindowManager.LayoutParams.MATCH_PARENT,
            WindowManager.LayoutParams.MATCH_PARENT,
            getOverlayWindowType(),
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                    WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE or
                    WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                    WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.LEFT
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }
            alpha = 0.78f
        }

        try {
            windowManager.addView(canvasOverlayView, canvasLayoutParams)
        } catch (e: Exception) {
            e.printStackTrace()
        }

        applyTableBoundedWindow()
    }

    /**
     * Table-Bounded Window Cropping:
     * Crops CanvasOverlayView window to table bounds (+ 18dp padding) during normal play
     * to keep Power Bar (left) and UI completely clear of overlay window interception.
     * Expands to MATCH_PARENT fullscreen when calibration mode is active.
     */
    private fun applyTableBoundedWindow() {
        val canvas = canvasOverlayView ?: return
        val density = resources.displayMetrics.density

        if (isCalibrationActive) {
            // Saat kalibrasi aktif, gunakan layar penuh agar pengguna bebas menarik 8 titik kalibrasi
            canvasLayoutParams.width = WindowManager.LayoutParams.MATCH_PARENT
            canvasLayoutParams.height = WindowManager.LayoutParams.MATCH_PARENT
            canvasLayoutParams.x = 0
            canvasLayoutParams.y = 0
            canvasLayoutParams.alpha = 1.0f
            canvasLayoutParams.gravity = Gravity.TOP or Gravity.LEFT
            canvasLayoutParams.flags = WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                    WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                    WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                canvasLayoutParams.layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }
            canvas.setWindowOffset(0f, 0f)
        } else {
            // Saat bermain normal, pangkas jendela kanvas HANYA seukuran kotak meja + padding 18dp
            // Area Power Bar di sisi kiri layar (x < winX) 100% bersih tanpa overlay!
            val pad = 18f * density
            val box = canvas.boundaryBox
            val winX = (box.left - pad).roundToInt().coerceAtLeast(0)
            val winY = (box.top - pad).roundToInt().coerceAtLeast(0)
            val winW = (box.width() + pad * 2f).roundToInt().coerceAtLeast(100)
            val winH = (box.height() + pad * 2f).roundToInt().coerceAtLeast(100)

            canvasLayoutParams.width = winW
            canvasLayoutParams.height = winH
            canvasLayoutParams.x = winX
            canvasLayoutParams.y = winY
            canvasLayoutParams.alpha = 0.78f
            canvasLayoutParams.gravity = Gravity.TOP or Gravity.LEFT
            canvasLayoutParams.flags = WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                    WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE or
                    WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                    WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                canvasLayoutParams.layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }

            canvas.setWindowOffset(winX.toFloat(), winY.toFloat())
        }

        try {
            windowManager.updateViewLayout(canvas, canvasLayoutParams)
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    /**
     * Calibration Mode: Enable touch on canvas and hide handles.
     */
    private fun setCalibrationMode(enabled: Boolean) {
        val canvas = canvasOverlayView ?: return
        isCalibrationActive = enabled

        if (enabled) {
            canvas.setCalibrating(true)
            syncHandlesVisibility()

            btnCalibrate?.text = getString(R.string.control_save_bound)
            btnCalibrate?.setBackgroundResource(R.drawable.bg_control_btn_active)
            btnCalibrate?.setTextColor(ContextCompat.getColor(this, R.color.green_status))
        } else {
            canvas.setCalibrating(false)
            syncHandlesVisibility()

            btnCalibrate?.text = getString(R.string.control_calibrate)
            btnCalibrate?.setBackgroundResource(R.drawable.bg_control_btn)
            btnCalibrate?.setTextColor(Color.WHITE)

            Toast.makeText(this, R.string.toast_calib_saved, Toast.LENGTH_SHORT).show()
        }

        applyTableBoundedWindow()
    }

    /**
     * Feature 2: Anti-Finger Obstruction (Offset Touch Handles).
     * The grip pad is positioned below the actual target ball to keep the ball visible.
     */
    private fun setupFloatingHandles() {
        val realSize = getRealScreenSize()
        val screenW = kotlin.math.max(realSize.x, realSize.y).toFloat()
        val screenH = kotlin.math.min(realSize.x, realSize.y).toFloat()

        val (ptA, ptB, ptC) = prefsManager.getPoints(screenW, screenH)
        val initialAlpha = prefsManager.getLineAlpha()
        val initialColor = prefsManager.getLineColor()

        // 1. Handle A
        handleViewA = HandleView(this, "A").apply {
            updateColor(initialColor)
            updateHandleAlpha(initialAlpha)
        }
        val gripPosA = calculateGripFromAim(ptA, isOffsetInvertedA)
        handleParamsA = createHandleLayoutParams(gripPosA)
        setupHandleTouchListener(
            handleViewA!!,
            handleParamsA,
            initialInverted = isOffsetInvertedA,
            onInversionChanged = { isOffsetInvertedA = it }
        ) { aimX, aimY, gripX, gripY ->
            canvasOverlayView?.setPointA(aimX, aimY, gripX, gripY)
        }
        windowManager.addView(handleViewA, handleParamsA)
        canvasOverlayView?.setPointA(ptA.x, ptA.y, gripPosA.x, gripPosA.y)

        // 2. Handle B (Target Ball)
        handleViewB = HandleView(this, "B").apply {
            updateColor(initialColor)
            updateHandleAlpha(initialAlpha)
        }
        val gripPosB = calculateGripFromAim(ptB, isOffsetInvertedB)
        handleParamsB = createHandleLayoutParams(gripPosB)
        setupHandleTouchListener(
            handleViewB!!,
            handleParamsB,
            initialInverted = isOffsetInvertedB,
            onInversionChanged = { isOffsetInvertedB = it }
        ) { aimX, aimY, gripX, gripY ->
            canvasOverlayView?.setPointB(aimX, aimY, gripX, gripY)
        }
        windowManager.addView(handleViewB, handleParamsB)
        canvasOverlayView?.setPointB(ptB.x, ptB.y, gripPosB.x, gripPosB.y)

        // 3. Handle C
        handleViewC = HandleView(this, "C").apply {
            updateColor(initialColor)
            updateHandleAlpha(initialAlpha)
        }
        val gripPosC = calculateGripFromAim(ptC, isOffsetInvertedC)
        handleParamsC = createHandleLayoutParams(gripPosC)
        setupHandleTouchListener(
            handleViewC!!,
            handleParamsC,
            initialInverted = isOffsetInvertedC,
            onInversionChanged = { isOffsetInvertedC = it }
        ) { aimX, aimY, gripX, gripY ->
            canvasOverlayView?.setPointC(aimX, aimY, gripX, gripY)
        }
        windowManager.addView(handleViewC, handleParamsC)
        canvasOverlayView?.setPointC(ptC.x, ptC.y, gripPosC.x, gripPosC.y)

        syncHandlesVisibility()
    }

    private fun calculateGripFromAim(aim: PointF, isInverted: Boolean = false): PointF {
        val realSize = getRealScreenSize()
        val currentScreenW = kotlin.math.max(realSize.x, realSize.y).toFloat()
        val currentScreenH = kotlin.math.min(realSize.x, realSize.y).toFloat()

        val gripX = (aim.x + handleOffsetX).coerceIn(handleSizePx / 2f, currentScreenW - handleSizePx / 2f)
        val targetGripY = if (isInverted) aim.y - handleOffsetY else aim.y + handleOffsetY
        val gripY = if (targetGripY + handleSizePx > currentScreenH - (20f * resources.displayMetrics.density)) {
            aim.y - handleOffsetY
        } else {
            targetGripY
        }.coerceIn(0f, currentScreenH)
        return PointF(gripX, gripY)
    }

    private fun createHandleLayoutParams(center: PointF): WindowManager.LayoutParams {
        return WindowManager.LayoutParams(
            handleSizePx,
            handleSizePx,
            getOverlayWindowType(),
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                    WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                    WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.LEFT
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }
            x = (center.x - handleSizePx / 2f).roundToInt()
            y = (center.y - handleSizePx / 2f).roundToInt()
        }
    }

    @SuppressLint("ClickableViewAccessibility")
    private fun setupHandleTouchListener(
        view: View,
        params: WindowManager.LayoutParams,
        initialInverted: Boolean = false,
        onInversionChanged: ((Boolean) -> Unit)? = null,
        onMoved: (aimX: Float, aimY: Float, gripX: Float, gripY: Float) -> Unit
    ) {
        var isOffsetInverted = initialInverted

        // Touch Tracking & Velocity Sensitivity
        var lastRawX = 0f
        var lastRawY = 0f
        var currentGripX = 0f
        var currentGripY = 0f

        var downTime = 0L
        var downRawX = 0f
        var downRawY = 0f

        // Long-Press Tracking (Smart Offset Flip)
        var isLongPressed = false
        var hasMovedSignificantly = false

        fun flipOffset() {
            val newInverted = !isOffsetInverted
            isOffsetInverted = newInverted
            onInversionChanged?.invoke(newInverted)

            val realSize = getRealScreenSize()
            val currentScreenW = kotlin.math.max(realSize.x, realSize.y).toFloat()
            val currentScreenH = kotlin.math.min(realSize.x, realSize.y).toFloat()

            // Invert grip Y: if newInverted, grip jumps above ball (-2 * handleOffsetY)
            // if !newInverted, grip jumps below ball (+2 * handleOffsetY)
            val deltaY = if (newInverted) -2f * handleOffsetY else 2f * handleOffsetY
            currentGripY = (currentGripY + deltaY).coerceIn(0f, currentScreenH)

            params.y = (currentGripY - handleSizePx / 2f).roundToInt()
            try {
                windowManager.updateViewLayout(view, params)
            } catch (e: Exception) {
                e.printStackTrace()
            }

            view.performHapticFeedback(HapticFeedbackConstants.LONG_PRESS)

            val aimX = (currentGripX - handleOffsetX).coerceIn(0f, currentScreenW)
            val offsetDir = if (isOffsetInverted) handleOffsetY else -handleOffsetY
            val aimY = (currentGripY + offsetDir).coerceIn(0f, currentScreenH)
            onMoved(aimX, aimY, currentGripX, currentGripY)
        }

        val longPressRunnable = Runnable {
            if (!hasMovedSignificantly) {
                isLongPressed = true
                flipOffset()
            }
        }

        view.setOnTouchListener { _, event ->
            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    lastRawX = event.rawX
                    lastRawY = event.rawY
                    downRawX = event.rawX
                    downRawY = event.rawY
                    downTime = System.currentTimeMillis()

                    currentGripX = params.x + (handleSizePx / 2f)
                    currentGripY = params.y + (handleSizePx / 2f)

                    hasMovedSignificantly = false
                    isLongPressed = false
                    mainHandler.postDelayed(longPressRunnable, 450L)
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    val frameDx = event.rawX - lastRawX
                    val frameDy = event.rawY - lastRawY
                    lastRawX = event.rawX
                    lastRawY = event.rawY

                    val totalMoveFromDown = hypot(event.rawX - downRawX, event.rawY - downRawY)
                    if (totalMoveFromDown > 8f * resources.displayMetrics.density) {
                        hasMovedSignificantly = true
                        mainHandler.removeCallbacks(longPressRunnable)
                    }

                    val speed = hypot(frameDx, frameDy)
                    // Deadzone Anti-Jitter: ignore micro movements during rest or lift-off
                    if (speed < 0.4f) {
                        return@setOnTouchListener true
                    }

                    // Dynamic Sensitivity Multiplier (Sniper Fine-Tuning Mode)
                    val sensitivity = when {
                        speed < 3.5f -> 0.28f  // Sniper Mode: ultra-fine alignment
                        speed < 10f -> 0.65f   // Medium adjustments
                        else -> 1.0f           // Fast panning
                    }

                    currentGripX += frameDx * sensitivity
                    currentGripY += frameDy * sensitivity

                    val realSize = getRealScreenSize()
                    val currentScreenW = kotlin.math.max(realSize.x, realSize.y).toFloat()
                    val currentScreenH = kotlin.math.min(realSize.x, realSize.y).toFloat()

                    // Auto-Flip when dragged past bottom limit and handle is below ball
                    if (!isOffsetInverted && currentGripY > currentScreenH - (handleSizePx * 0.8f)) {
                        flipOffset()
                    }

                    currentGripX = currentGripX.coerceIn(0f, currentScreenW)
                    currentGripY = currentGripY.coerceIn(0f, currentScreenH)

                    params.x = (currentGripX - handleSizePx / 2f).roundToInt()
                    params.y = (currentGripY - handleSizePx / 2f).roundToInt()

                    try {
                        windowManager.updateViewLayout(view, params)
                    } catch (e: Exception) {
                        e.printStackTrace()
                    }

                    val aimX = (currentGripX - handleOffsetX).coerceIn(0f, currentScreenW)
                    val offsetDir = if (isOffsetInverted) handleOffsetY else -handleOffsetY
                    val aimY = (currentGripY + offsetDir).coerceIn(0f, currentScreenH)

                    onMoved(aimX, aimY, currentGripX, currentGripY)
                    true
                }
                MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                    mainHandler.removeCallbacks(longPressRunnable)
                    isLongPressed = false
                    persistHandlePositions()
                    true
                }
                else -> false
            }
        }
    }

    private fun persistHandlePositions() {
        val canvas = canvasOverlayView ?: return
        prefsManager.savePoints(canvas.pointA, canvas.pointB, canvas.pointC)
    }

    @SuppressLint("ClickableViewAccessibility")
    private fun setupFloatingControl() {
        val inflater = LayoutInflater.from(this)
        val controlView = inflater.inflate(R.layout.layout_floating_control, null)
        floatingControlView = controlView

        controlLayoutParams = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            getOverlayWindowType(),
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                    WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                    WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.LEFT
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
                layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            }
            x = 40
            y = 140
        }

        // References to UI components
        fullToolbarContainer = controlView.findViewById(R.id.fullToolbarContainer)
        minimizedCapsule = controlView.findViewById(R.id.minimizedCapsule)
        btnMiniDrag = controlView.findViewById(R.id.btnMiniDrag)
        btnMiniToggleOnOff = controlView.findViewById(R.id.btnMiniToggleOnOff)
        btnMiniExpand = controlView.findViewById(R.id.btnMiniExpand)

        val btnDrag = controlView.findViewById<TextView>(R.id.btnDragHandle)
        btnCalibrate = controlView.findViewById(R.id.btnToggleCalibrate)
        btnSettings = controlView.findViewById(R.id.btnToggleSettings)
        val btnMinimize = controlView.findViewById<TextView>(R.id.btnMinimize)
        val btnClose = controlView.findViewById<TextView>(R.id.btnCloseOverlay)

        panelSettings = controlView.findViewById(R.id.panelSettings)
        val tvThickness = controlView.findViewById<TextView>(R.id.tvThicknessValue)
        val seekThickness = controlView.findViewById<SeekBar>(R.id.seekThickness)
        val tvOpacity = controlView.findViewById<TextView>(R.id.tvOpacityValue)
        val seekOpacity = controlView.findViewById<SeekBar>(R.id.seekOpacity)
        val tvBallSize = controlView.findViewById<TextView>(R.id.tvBallSizeValue)
        val seekBallSize = controlView.findViewById<SeekBar>(R.id.seekBallSize)

        // 1. Dragging toolbar
        setupDraggableBar(btnDrag, controlView)

        // Smart Mini Bubble (Drag to move, Tap to expand, Long-press to toggle ON/OFF)
        btnMiniExpand?.let { setupSmartMiniBubble(it, controlView) }

        // 2. Calibration Mode Toggle
        btnCalibrate?.setOnClickListener {
            if (isCanvasHidden) {
                toggleCanvasVisibility()
            }
            setCalibrationMode(!isCalibrationActive)
        }

        // 7. Settings Sub-Panel (Thickness & Opacity Sliders)
        val initialStroke = prefsManager.getLineStrokeWidth()
        val initialAlpha = prefsManager.getLineAlpha()

        seekThickness.progress = (initialStroke / 0.5f).roundToInt().coerceIn(1, 30)
        seekOpacity.progress = initialAlpha.coerceIn(25, 255)

        tvThickness.text = getString(R.string.label_thickness, initialStroke)
        val initialPct = (initialAlpha * 100 / 255)
        tvOpacity.text = getString(R.string.label_opacity, initialPct)

        btnSettings?.setOnClickListener {
            val panel = panelSettings ?: return@setOnClickListener
            val isPanelOpen = panel.visibility == View.VISIBLE
            if (isPanelOpen) {
                panel.visibility = View.GONE
                btnSettings?.setBackgroundResource(R.drawable.bg_control_btn)
            } else {
                panel.visibility = View.VISIBLE
                btnSettings?.setBackgroundResource(R.drawable.bg_control_btn_active)
            }
        }

        seekThickness.setOnSeekBarChangeListener(object : SeekBar.OnSeekBarChangeListener {
            override fun onProgressChanged(seekBar: SeekBar?, progress: Int, fromUser: Boolean) {
                val strokeWidth = (progress.coerceAtLeast(1) * 0.5f)
                tvThickness.text = getString(R.string.label_thickness, strokeWidth)
                val alpha = seekOpacity.progress
                canvasOverlayView?.updateLineStyle(strokeWidth, alpha)
            }

            override fun onStartTrackingTouch(seekBar: SeekBar?) {}

            override fun onStopTrackingTouch(seekBar: SeekBar?) {
                val progress = seekBar?.progress ?: 7
                val strokeWidth = (progress.coerceAtLeast(1) * 0.5f)
                val alpha = seekOpacity.progress
                prefsManager.saveLineStyle(strokeWidth, alpha)
            }
        })

        seekOpacity.setOnSeekBarChangeListener(object : SeekBar.OnSeekBarChangeListener {
            override fun onProgressChanged(seekBar: SeekBar?, progress: Int, fromUser: Boolean) {
                val alpha = progress.coerceIn(25, 255)
                val pct = (alpha * 100 / 255)
                tvOpacity.text = getString(R.string.label_opacity, pct)
                val strokeWidth = (seekThickness.progress.coerceAtLeast(1) * 0.5f)
                canvasOverlayView?.updateLineStyle(strokeWidth, alpha)
                updateAllHandlesAlpha(alpha)
            }

            override fun onStartTrackingTouch(seekBar: SeekBar?) {}

            override fun onStopTrackingTouch(seekBar: SeekBar?) {
                val alpha = (seekBar?.progress ?: 220).coerceIn(25, 255)
                val strokeWidth = (seekThickness.progress.coerceAtLeast(1) * 0.5f)
                prefsManager.saveLineStyle(strokeWidth, alpha)
                updateAllHandlesAlpha(alpha)
            }
        })

        // Ball Ring Size Slider Setup
        val initialBallRadius = prefsManager.getRingRadius()
        seekBallSize.progress = initialBallRadius.roundToInt().coerceIn(8, 30)
        tvBallSize.text = getString(R.string.label_ball_size, initialBallRadius)

        seekBallSize.setOnSeekBarChangeListener(object : SeekBar.OnSeekBarChangeListener {
            override fun onProgressChanged(seekBar: SeekBar?, progress: Int, fromUser: Boolean) {
                val radiusDp = progress.coerceIn(8, 30) * 1.0f
                tvBallSize.text = getString(R.string.label_ball_size, radiusDp)
                if (fromUser) {
                    canvasOverlayView?.setRingRadius(radiusDp)
                }
            }

            override fun onStartTrackingTouch(seekBar: SeekBar?) {}

            override fun onStopTrackingTouch(seekBar: SeekBar?) {
                val progress = seekBar?.progress ?: 14
                val radiusDp = progress.coerceIn(8, 30) * 1.0f
                prefsManager.saveRingRadius(radiusDp)
                canvasOverlayView?.setRingRadius(radiusDp)
            }
        })

        // 8. Feature 2: Line Color Picker (Preset Swatches & Hue Spectrum Slider)
        val density = resources.displayMetrics.density

                    val dy = (event.rawY - initialTouchY).toInt()
                    controlLayoutParams.x = initialX + dx

                    true

    @SuppressLint("ClickableViewAccessibility")
    private fun setupSmartMiniBubble(bubbleView: TextView, rootView: View) {
        var initialX = 0
        var initialY = 0
        var downRawX = 0f
        var downRawY = 0f
        var isDragging = false
        var isLongPressTriggered = false
        val touchSlop = 8f * resources.displayMetrics.density

        val longPressRunnable = Runnable {
            if (!isDragging) {
                isLongPressTriggered = true
                bubbleView.performHapticFeedback(android.view.HapticFeedbackConstants.LONG_PRESS)
                toggleCanvasVisibility()
            }
        }

        bubbleView.setOnTouchListener { _, event ->
            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    initialX = controlLayoutParams.x
                    initialY = controlLayoutParams.y
                    downRawX = event.rawX
                    downRawY = event.rawY
                    isDragging = false
                    isLongPressTriggered = false
                    mainHandler.postDelayed(longPressRunnable, 400L)
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    val dx = event.rawX - downRawX
                    val dy = event.rawY - downRawY
                    if (kotlin.math.hypot(dx, dy) > touchSlop) {
                        isDragging = true
                        mainHandler.removeCallbacks(longPressRunnable)
                        controlLayoutParams.x = initialX + dx.roundToInt()
                        controlLayoutParams.y = initialY + dy.roundToInt()
                        try {
                            windowManager.updateViewLayout(rootView, controlLayoutParams)
                        } catch (e: Exception) {
                            e.printStackTrace()
                        }
                    }
                    true
                }
                MotionEvent.ACTION_UP -> {
                    mainHandler.removeCallbacks(longPressRunnable)
                    if (!isDragging && !isLongPressTriggered) {
                        // Single Tap: Buka kembali toolbar penuh
                        expandToolbar()
                    }
                    true
                }
                MotionEvent.ACTION_CANCEL -> {
                    mainHandler.removeCallbacks(longPressRunnable)
                    true
                }
                else -> false
            }
        }
    }

class PrefsManager(context: Context) {

    /**
     * Save the 4 calibrated boundary edges.
     */
    fun saveBoundaryBox(left: Float, top: Float, right: Float, bottom: Float) {
        prefs.edit().apply {
            putFloat(KEY_BOUND_LEFT, left)
            putFloat(KEY_BOUND_TOP, top)
            putFloat(KEY_BOUND_RIGHT, right)
            putFloat(KEY_BOUND_BOTTOM, bottom)
            putBoolean(KEY_BOUND_CALIBRATED, true)
            putInt(KEY_CALIB_VERSION, CURRENT_CALIB_VERSION)
            apply()
        }
    }

    /**
     * Retrieve the calibrated boundary box. If not calibrated yet or saved by an older
     * version with cutout skew, computes default margins matching standard 8 Ball Pool table.
     */
    fun getBoundaryBox(screenWidth: Float, screenHeight: Float): RectF {
        val landW = kotlin.math.max(screenWidth, screenHeight)
        val landH = kotlin.math.min(screenWidth, screenHeight)

        val calibVersion = prefs.getInt(KEY_CALIB_VERSION, 0)
        val isCalibrated = prefs.getBoolean(KEY_BOUND_CALIBRATED, false)

        if (isCalibrated && calibVersion >= CURRENT_CALIB_VERSION) {
            val left = prefs.getFloat(KEY_BOUND_LEFT, landW * 0.191f)
            val top = prefs.getFloat(KEY_BOUND_TOP, landH * 0.228f)
            val right = prefs.getFloat(KEY_BOUND_RIGHT, landW * (1f - 0.191f))
            val bottom = prefs.getFloat(KEY_BOUND_BOTTOM, landH * 0.921f)

            // Sanity check to avoid inverted or collapsed rectangles and auto-clamp to screen
            val minMargin = 20f
            val clampedL = left.coerceIn(minMargin, landW * 0.45f)
            val clampedT = top.coerceIn(minMargin, landH * 0.45f)
            val clampedR = right.coerceIn(landW * 0.55f, landW - minMargin)
            val clampedB = bottom.coerceIn(landH * 0.55f, landH - minMargin)

            if (clampedR > clampedL + 50f && clampedB > clampedT + 50f) {
                return RectF(clampedL, clampedT, clampedR, clampedB)
            }
        }

        // If calibration was from an older version with cutout offset, reset it
        if (isCalibrated && calibVersion < CURRENT_CALIB_VERSION) {
            resetBoundaryBox()
        }

        // Default bounds: 8 Ball Pool table proportions (2:1 playing surface centered horizontally)
        val insetX = (landW * 0.191f).coerceAtLeast(40f)
        val topY = (landH * 0.228f).coerceAtLeast(40f)
        val botY = (landH * 0.921f).coerceAtLeast(topY + 100f)
        return RectF(insetX, topY, landW - insetX, botY)
    }
